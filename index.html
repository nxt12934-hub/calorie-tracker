// AI 拍照辨識食物 (自動匹配最新可用模型) async function handleImage(event) { const file = event.target.files[0]; if (!file) return; const apiKey = localStorage.getItem('gemini_api_key'); if (!apiKey) { alert('請先在下方展開設定並填入 Google Gemini API Key！'); return; } document.getElementById('imagePreview').src = URL.createObjectURL(file); document.getElementById('imagePreview').style.display = 'block'; const status = document.getElementById('statusText'); status.innerText = '壓縮圖片並由 AI 辨識中...'; try { const base64 = await resizeImageToBase64(file, 800); // 優先嘗試官方建議的最新主力模型 const candidateModels = [ 'gemini-2.5-flash', 'gemini-flash-latest', 'gemini-1.5-flash-latest' ]; let targetModel = candidateModels[0]; // 自動嘗試動態獲取當前帳號可用的 flash 模型清單 try { const listRes = await fetch(`https://generativelanguage.googleapis.com/v1beta/models?key=${apiKey.trim()}`); if (listRes.ok) { const listData = await listRes.json(); const availableFlash = listData.models?.filter(m => m.supportedGenerationMethods?.includes('generateContent') && m.name.includes('flash') ); if (availableFlash && availableFlash.length > 0) { // 取得清單中最新的一個模型 targetModel = availableFlash[availableFlash.length - 1].name.replace('models/', ''); } } } catch (e) { console.warn("無法取得模型清單，使用預設值：", e); } const prompt = `請辨識這張圖片中的食物、飲料或營養標示。若有營養標示請依每份熱量與包裝份數相乘計算總熱量(kcal)與蛋白質(g)；若無標示請估算整份食物數值。必須嚴格回傳標準 JSON 格式，不要加 markdown 或其他字元：{"foodName":"品名","calories":168,"protein":8.2}`; const url = `https://generativelanguage.googleapis.com/v1beta/models/${targetModel}:generateContent?key=${apiKey.trim()}`; const res = await fetch(url, { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify({ contents: [{ parts: [ { text: prompt }, { inline_data: { mime_type: "image/jpeg", data: base64 } } ] }] }) }); const data = await res.json(); if (!res.ok || data.error) { throw new Error(data.error ? data.error.message : `HTTP ${res.status}`); } const candidate = data.candidates?.[0]; if (!candidate?.content?.parts?.[0]?.text) { throw new Error('AI 回傳格式不符合預期'); } const raw = candidate.content.parts[0].text.replace(/```json/g, '').replace(/```/g, '').trim(); const parsed = JSON.parse(raw); fillForm(parsed.foodName, parsed.calories, parsed.protein); status.innerText = `辨識完成：${parsed.foodName} (使用 ${targetModel})`; } catch (e) { console.error("辨識詳細錯誤：", e); status.innerText = '辨識失敗：' + e.message; } }
